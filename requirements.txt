# PikaPet runtime, macOS and Windows. Do not install this by hand -- run the
# installer for your platform, which also checks the Python version and the
# asset links:
#
#   macOS    ./install.sh
#   Windows  powershell -ExecutionPolicy Bypass -File install.ps1
#
# These are the versions the port was verified on. PikaPet.exe bundled slightly
# older ones (numpy 2.5.0, websockets 16.0) and runs fine on both.
#
# Only pillow, pystray and websockets are actually imported by the game
# (pet.pyc, spriteanim.pyc); the rest mirror what the .exe bundled. winlayer.pyc
# needs nothing beyond ctypes, so Windows has no extra requirement.
pillow==12.3.0
numpy==2.5.3
pystray==0.19.5
websockets==17.1
Werkzeug==3.1.8
MarkupSafe==3.0.3
watchdog==6.0.0
charset-normalizer==3.5.1
colorama==0.4.6
packaging==26.3
six==1.17.0
threadpoolctl==3.7.0
typing_extensions==4.16.0

# macOS only: run/maclayer.py calls Quartz/AppKit directly, and pystray's macOS
# backend pulls these in too. Marked so `pip install -r` works on Windows.
pyobjc-core==12.2.2; sys_platform == "darwin"
pyobjc-framework-Cocoa==12.2.2; sys_platform == "darwin"
pyobjc-framework-Quartz==12.2.2; sys_platform == "darwin"
# run/macnotify.py 가 쓴다. 이 앱 소유의 알림 배너를 띄우려면 모던
# UNUserNotificationCenter 가 필요하고, 그 블록 콜백에는 pyobjc 메타데이터가
# 있어야 한다. 없으면 osascript로 떨어져서 배너 소유자가 스크립트 편집기가 된다.
pyobjc-framework-UserNotifications==12.2.2; sys_platform == "darwin"
